import { Injectable } from '@angular/core';
import { HttpClient, HttpErrorResponse, HttpHeaders } from '@angular/common/http';
import { Observable, throwError } from 'rxjs';
import { catchError } from 'rxjs/operators';
import { AuthService } from './AuthService'; // Adjust the path if needed

/**
 * Interface for creating a new demande.
 * Mirrors the DemandeCreateDto from the C# backend.
 */
export interface DemandeCreateDto {
    annonceId: number;
    dureeLocationEnJours?: number;
}

/**
 * Interface representing the response after creating a demande.
 * Mirrors the DemandeCreatedDto from the C# backend.
 */
export interface DemandeCreatedDto {
    id: number;
    annonceId: number;
    demandeurId: number;
    statut: string;
    dateDemande: Date;
    dureeLocationEnJours?: number;

}

/**
 * Interface for a user summary.
 * Mirrors the UtilisateurSummaryDto from the C# backend.
 */
export interface UtilisateurSummaryDto {
    id: number;
    prenom: string;
    nom: string;
}

/**
 * Interface for an annonce summary.
 * Mirrors the AnnonceSummaryDto from the C# backend.
 */
export interface AnnonceSummaryDto {
    id: number;
    titre: string;
    type: string;
    prix: number;
    statut: string;
    utilisateur: UtilisateurSummaryDto;
}

/**
 * Interface for detailed demande information.
 * Mirrors the DemandeDetailsDto from the C# backend.
 */
export interface DemandeDetailsDto {
    id: number;
    annonce: AnnonceSummaryDto;
    demandeur: UtilisateurSummaryDto;
    statut: string;
    dateDemande: Date;
    dureeLocationEnJours?: number;
}
/**
 * Interface for a notification.
 * Mirrors the NotificationDto from your backend.
 */
export interface INotification {
    id: number;
    message: string;
    type: string;
    isRead: boolean;
    created_at: Date;
    demandeId?: number;
    annonceId?: number;
}
/**
 * Interface for listing demandes.
 * Mirrors the DemandeListDto from the C# backend.
 */
export interface DemandeListDto {
    id: number;
    annonceId: number;
    demandeur: UtilisateurSummaryDto;
    statut: string;
    dateDemande: Date;
    dureeLocationEnJours?: number;
}

@Injectable({
    providedIn: 'root'
})
export class DemandeService {
    private apiUrl = 'https://localhost:7272/api/Demande';

    constructor(private http: HttpClient, private authService: AuthService) {}

    /**
     * General error handling function for all HTTP requests.
     * Logs the error and re-throws a user-friendly message.
     * @param error The HttpErrorResponse.
     */
    private handleError(error: HttpErrorResponse): Observable<never> {
        let errorMessage = 'An unknown error occurred!';
        if (error.error instanceof ErrorEvent) {
            // Client-side or network error occurred.
            errorMessage = `Client Error: ${error.error.message}`;
        } else {
            // The backend returned an unsuccessful response code.
            const backendError = error.error;
            if (typeof backendError === 'string') {
                errorMessage = backendError;
            } else if (backendError && backendError.Message) {
                errorMessage = backendError.Message;
            } else {
                errorMessage = `Backend returned code ${error.status}: ${error.statusText || 'Unknown Error'}`;
            }
        }
        console.error(errorMessage);
        return throwError(() => new Error(errorMessage));
    }

    /**
     * Creates headers with the Authorization token.
     * @returns HttpHeaders with Bearer token or throws an error if token is missing.
     */
    private getAuthHeaders(): HttpHeaders {
        const token = this.authService.getToken();
        if (!token) {
            throw new Error('Authentication token not found. Please log in.');
        }
        return new HttpHeaders({
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json'
        });
    }

    /**
     * Creates a new demande.
     * @param demande The DemandeCreateDto data.
     * @returns An Observable of the created DemandeCreatedDto.
     */
    createDemande(demande: DemandeCreateDto): Observable<DemandeCreatedDto> {
        return this.http
            .post<DemandeCreatedDto>(this.apiUrl, demande, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves a specific demande by ID.
     * @param id The ID of the demande.
     * @returns An Observable of DemandeDetailsDto.
     */
    getDemande(id: number): Observable<DemandeDetailsDto> {
        return this.http
            .get<DemandeDetailsDto>(`${this.apiUrl}/${id}`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves the status of a specific demande by ID.
     * @param id The ID of the demande.
     * @returns An Observable of an object containing the statut.
     */
    getDemandeStatus(id: number): Observable<{ statut: string }> {
        return this.http
            .get<{ statut: string }>(`${this.apiUrl}/${id}/statut`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves all demandes for a specific annonce.
     * @param annonceId The ID of the annonce.
     * @returns An Observable of an array of DemandeListDto.
     */
    getDemandesParAnnonce(annonceId: number): Observable<DemandeListDto[]> {
        return this.http
            .get<DemandeListDto[]>(`${this.apiUrl}/annonce/${annonceId}`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves all demandes made by the authenticated user.
     * @returns An Observable of an array of DemandeDetailsDto.
     */
    getMesDemandes(): Observable<DemandeDetailsDto[]> {
        return this.http
            .get<DemandeDetailsDto[]>(`${this.apiUrl}/mes-demandes`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves all demandes for annonces owned by the authenticated user.
     * @returns An Observable of an array of DemandeDetailsDto.
     */
    getDemandesPourMesAnnonces(): Observable<DemandeDetailsDto[]> {
        return this.http
            .get<DemandeDetailsDto[]>(`${this.apiUrl}/demandes-pour-mes-annonces`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves all accepted demandes for the authenticated user.
     * @returns An Observable of an array of DemandeDetailsDto.
     */
    getAcceptedDemands(): Observable<DemandeDetailsDto[]> {
        return this.http
            .get<DemandeDetailsDto[]>(`${this.apiUrl}/accepted`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Retrieves all canceled demandes for the authenticated user.
     * @returns An Observable of an array of DemandeDetailsDto.
     */
    getCancelledDemands(): Observable<DemandeDetailsDto[]> {
        return this.http
            .get<DemandeDetailsDto[]>(`${this.apiUrl}/cancelled`, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Accepts a demande, creating an associated commande and transaction.
     * @param id The ID of the demande to accept.
     * @returns An Observable of void.
     */
    accepterDemande(id: number): Observable<void> {
        return this.http
            .put<void>(`${this.apiUrl}/${id}/AccepterDemande`, {}, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Refuses a demande.
     * @param id The ID of the demande to refuse.
     * @returns An Observable of void.
     */
    refuserDemande(id: number): Observable<void> {
        return this.http
            .put<void>(`${this.apiUrl}/${id}/refuser`, {}, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Cancels a demande.
     * @param id The ID of the demande to cancel.
     * @returns An Observable of void.
     */
    annulerDemande(id: number): Observable<void> {
        return this.http
            .put<void>(`${this.apiUrl}/${id}/annuler`, {}, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Revokes the refusal of a demande, setting it back to 'EnAttente'.
     * @param id The ID of the demande to revoke refusal for.
     * @returns An Observable of void.
     */
    revoquerRefusDemande(id: number): Observable<void> {
        return this.http
            .put<void>(`${this.apiUrl}/${id}/revoquer-refus`, {}, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }

    /**
     * Revokes the acceptance of a demande, setting it back to 'EnAttente' and removing associated commande.
     * @param id The ID of the demande to revoke acceptance for.
     * @returns An Observable of void.
     */
    revoquerAcceptationDemande(id: number): Observable<void> {
        return this.http
            .put<void>(`${this.apiUrl}/${id}/revoquer-acceptation`, {}, { headers: this.getAuthHeaders() })
            .pipe(catchError(this.handleError));
    }
}
